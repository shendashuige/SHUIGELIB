        return 0

        s = s[1:]

def levelOrder(root):
    ans = []
    if not root:
        return ans
    q = deque([root])
    while q:
        size = len(q)
        row = []
        for _ in range(size):
            node = q.popleft()
            row.append(node.val)
            if node.left: q.append(node.left)
            if node.right: q.append(node.right)
        ans.append(row)
    return ans

        if not node:
            return 0
        l = dfs(node.left)

def permute(nums):
    ans=[]
    def dfs(path,used):
        if len(path)==len(nums):
            ans.append(path[:])
            return
        for i in range(len(nums)):
            if not used[i]:
                used[i]=True
                path.append(nums[i])
                dfs(path,used)
                path.pop()
                used[i]=False
    dfs([],[False]*len(nums))
    return ans

        if i<0 or i>=m or j<0 or j>=n or grid[i][j]=="0":

    if not nums: return 0
    if len(nums)==1: return nums[0]
    dp=[0]*len(nums)
    dp[0]=nums[0]
    dp[1]=max(nums[0],nums[1])

    return dp[-1]

    dp=[INF]*(amount+1)

            if i>0: dp[i][j] += dp[i-1][j]
            if j>0: dp[i][j] += dp[i][j-1]
    return dp[-1][-1]

def mergeSort(arr):
    if len(arr)<=1:
        return arr
    mid = len(arr)//2
    L = mergeSort(arr[:mid])
    R = mergeSort(arr[mid:])
    i=j=0
    res=[]

        if L[i]<R[j]:
            res.append(L[i])
            i+=1
        else:
            res.append(R[j])
            j+=1
    res.extend(L[i:])
    res.extend(R[j:])
    return res

def heapSort(arr):
    def heapify(a,n,i):
        largest=i
        l=2*i+1
        r=2*i+2
        if l<n and a[l]>a[largest]: largest=l
        if r<n and a[r]>a[largest]: largest=r
        if largest!=i:
            a[i],a[largest]=a[largest],a[i]
            heapify(a,n,largest)
    n=len(arr)
    for i in range(n//2-1,-1,-1):
        heapify(arr,n,i)
    for i in range(n-1,0,-1):
        arr[0],arr[i]=arr[i],arr[0]
        heapify(arr,i,0)
    return arr

def longestPalindrome(s):
    def expand(l,r):
        while l>=0 and r<len(s) and s[l]==s[r]:
            l -=1
            r +=1
        return s[l+1:r]
    best=""
    for i in range(len(s)):
        a = expand(i,i)
        b = expand(i,i+1)
        best = max(best,a,b,key=len)
    return best
100. LRU 缓存（哈希 + 双向链表）

class Node:
    def __init__(self,k,v):
        self.key=k
        self.val=v
        self.prev=None
        self.next=None
class LRUCache:
    def __init__(self, capacity: int):
        self.cap = capacity
        self.d = {}
        self.head = Node(0,0)
        self.tail = Node(0,0)
        self.head.next = self.tail
        self.tail.prev = self.head
    def _remove(self,node):
        p,n = node.prev,node.next
        p.next = n
        n.prev = p
    def _add(self,node):
        node.prev = self.head
        node.next = self.head.next
        self.head.next.prev = node
        self.head.next = node
    def _moveToHead(self,node):
        self._remove(node)
        self._add(node)
    def get(self, key: int) -> int:
        if key not in self.d:
            return -1
        node = self.d[key]
        self._moveToHead(node)
        return node.val
    def put(self, key: int, value: int) -> None:
        if key in self.d:
            node = self.d[key]
            node.val = value
            self._moveToHead(node)
        else:
            newNode = Node(key,value)
            self.d[key] = newNode
            self._add(newNode)
            if len(self.d) > self.cap:
                delNode = self.tail.prev
                self._remove(delNode)
                del self.d[delNode.key]
